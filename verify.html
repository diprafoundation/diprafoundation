function verifyFreeCert(certId){ try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('FreeCertificates'); if(!sheet) return { status: 'error', message: 'Sheet not found' }; const data = sheet.getDataRange().getValues(); if(data.length < 2) return { status: 'error', message: 'No certificates' }; const headers = data[0]; const certIdCol = headers.indexOf('Certificate ID'); if(certIdCol === -1) return { status: 'error', message: 'Certificate ID column missing' }; for(let i = 1; i < data.length; i++){ const row = data[i]; if(String(row[certIdCol]).trim() === String(certId).trim()){ const record = {}; headers.forEach((h, idx) => { record[h] = row[idx]; // Also add camelCase version for compatibility const camel = String(h).replace(/\s+/g, '').replace(/^(.)/, (m) => m.toLowerCase()); record[camel] = row[idx]; }); return { status: 'success', data: record }; } } return { status: 'error', message: 'Certificate not found' }; } catch(e){ return { status: 'error', message: e.toString() }; } }
